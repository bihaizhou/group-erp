export function recordDraft(row: ExecutionRecord): Draft {
  const value = {
    ...blankDraft(row.subjectId, row.occurredOn),
    ...Object.fromEntries(
      [
        'warehouseId',
        'destinationWarehouseId',
        'partnerId',
        'currency',
        'deliveryOn',
        'deliveryTerms',
        'reason',
        'notes',
        'kind',
        'originalPostingId',
        'initializationReference',
        'productionEvidence',
        'lossEvidence',
        'inspectionEvidence',
        'evidence',
      ]
        .filter((key) => row[key as keyof ExecutionRecord] != null)
        .map((key) => [key, row[key as keyof ExecutionRecord]]),
    ),
    lines: row.lines.map((line) =>
      line.baselineQuantity !== undefined
        ? {
            id: line.id,
            skuId: line.skuId,
            quality: line.quality,
            actualQuantity: line.actualQuantity ?? null,
            reason: line.reason ?? '',
            notes: line.notes,
          }
        : structuredClone(line.selectionInput),
    ),
  };
  if (row.kind) {
    const source = row[sourceField[row.kind] as keyof ExecutionRecord];
    value.sourceId = typeof source === 'string' ? source : '';
  }
  return value;
}

export function validateDraft(
  page: ExecutionPage,
  draft: Draft,
  changeReason?: string,
) {
  const errors: Record<string, string> = {};
  for (const key of ['subjectId', 'warehouseId', 'occurredOn'] as const)
    if (!draft[key]) errors[key] = 'required';
  if (isOrder(page)) {
    if (!draft.partnerId) errors.partnerId = 'required';
    if (!/^[A-Za-z]{3}$/u.test(draft.currency))
      errors.currency = 'currencyInvalid';
  }
  if (
    page.kind === 'stockTransfers' &&
    (!draft.destinationWarehouseId ||
      draft.destinationWarehouseId === draft.warehouseId)
  )
    errors.destinationWarehouseId = 'differentWarehouse';
  if (isWarehouse(page)) {
    if (!draft.reason.trim()) errors.reason = 'required';
    if (sourceField[draft.kind] && !draft.sourceId)
      errors.sourceId = 'required';
    for (const field of [
      'initializationReference',
      'lossEvidence',
      'inspectionEvidence',
      'evidence',
    ] as const)
      if (
        ({
          initializationReference: 'initial',
          lossEvidence: 'loss',
          inspectionEvidence: 'qualityTransfer',
          evidence: 'reversal',
        }[field] === draft.kind ||
          (field === 'evidence' && draft.kind === 'transferDiscrepancy')) &&
        !draft[field].trim()
      )
        errors[field] = 'required';
    if (['productionReceipt', 'productionIssue'].includes(draft.kind))
      for (const field of [
        'reference',
        'title',
        'description',
        'occurredOn',
      ] as const)
        if (!draft.productionEvidence[field].trim())
          errors[`productionEvidence.${field}`] = 'required';
  }
  if (changeReason !== undefined && !changeReason.trim())
    errors.changeReason = 'required';
  if (
    page.kind !== 'stockCounts' &&
    draft.kind !== 'reversal' &&
    !draft.lines.length
  )
    errors.lines = 'linesRequired';
  for (const line of draft.lines) {
    const prefix = `line-${line.id}-`,
      quantity =
        page.kind === 'stockCounts'
          ? line.actualQuantity
          : (line.selection?.quantity ?? line.quantity);
    if (page.kind === 'stockCounts' && quantity === null) continue;
    if (
      !quantity ||
      !/^(0|[1-9]\d{0,17})(\.\d{1,6})?$/u.test(quantity) ||
      (page.kind !== 'stockCounts' && !/[1-9]/u.test(quantity))
    )
      errors[`${prefix}quantity`] = 'quantityInvalid';
    if (line.selection && (!line.selection.skuId || !line.selection.unitId))
      errors[`${prefix}selection`] = 'required';
    if (
      page.kind === 'stockCounts' &&
      line.actualQuantity !== null &&
      !line.skuId
    )
      errors[`${prefix}selection`] = 'required';
    if (
      ['salesReturn', 'issueReturn'].includes(draft.kind) &&
      !line.condition?.trim()
    )
      errors[`${prefix}condition`] = 'required';
  }
  return errors;
}

### “谁领的”实际查到什么程度

系统里有完整的申请侧字段：`requesterEmployeeId`、`recipientEmployeeId`，出库单也有关联的 `issueRequestId` / `issueRequestLineId`。但链路在出库这一步断了：

- 出库单详情只投影公司、仓库、供应商，不显示领用人；仓库操作员选源单时只看到申请单号，看不到“这批货给谁”。
- 领用申请列表的 `mine` 只查创建人和申请人，**领用人自己看不到待领清单**，也没有“确认收到”动作。
- 申请单状态只有 `draft/pending/approved/cancelled`，没有 `issued/partial/completed`；部分出库后申请仍是 approved，甚至可以取消。
- `stockIssueFulfillments` 表记录了已发/已退数量，但申请 API、详情页、运营总览都没有消费它。

所以现在只能回答“申请批给谁”，不能回答“实际谁在什么时候领走、领了多少、有没有签收异议”。

### 数量引擎本身是扎实的

底层过账并不是最弱的地方：`stockPostings` 一单唯一、`stockMovements` 不可变、负库存校验、预留、在途、冲销、审计、请求幂等都在。真正的问题是**业务控制点**，不是库存算术。

### 关键证据

- 拒收数量：`tests/logic/warehouse-execution.test.ts:166` 设置拒收 1，仍断言库存 6、订单已收 6；过账分支 [warehouse-posting.ts](group-erp/server/providers/warehouse-posting.ts#L313) 直接按 `baseQuantity` 计。
- 单人过账：受复核控制的只有 4 类，见 [warehouse-service.ts](group-erp/server/providers/warehouse-service.ts#L41)；其余类型在 [warehouse-service.ts](group-erp/server/providers/warehouse-service.ts#L514) 直接拒绝提交复核，允许从草稿过账。
- 领用状态：状态枚举只有四个，见 [issue-request-model.ts](group-erp/server/providers/issue-request-model.ts#L8)；`mine` 过滤见 [issue-request-service.ts](group-erp/server/providers/issue-request-service.ts#L354)；取消只检查预留，不看已发数量，见 [issue-request-service.ts](group-erp/server/providers/issue-request-service.ts#L1657)。
- 订单不结案：订单状态只有 `draft/pending/approved/cancelled`，见 [execution-workflow.ts](group-erp/server/providers/execution-workflow.ts#L81)；运营仍把 approved 当未结，见 [operations-service.ts](group-erp/server/providers/operations-service.ts#L338)；取消只检查预留，见 [trade-order-service.ts](group-erp/server/providers/trade-order-service.ts#L772)。
- 库存维度：只有仓库、SKU、质量，见 [inventory_ledger.ts](group-erp/database/main/migrations/202610190001_inventory_ledger.ts#L426)，没有批次、库位、效期、成本。
- 固定资产：全仓没有资产表/资产 API；物料只有 `stock`/`nonStock` 两种管理模式，见 [material-model.ts](group-erp/server/providers/material-model.ts#L85)。非库存采购只会完成订单，不会生成资产卡。

### 建议先补的顺序

1. **P0 拒收扣减**：收货过账时把 `rejectedQuantity` 从入库数量和订单已收数量中扣掉，并明确拒收货物是退回供应商还是进待检库存。
2. **P0 领用闭环**：申请单增加已发/未发数量和“已完成”状态；出库单显示领用人；领用人可确认签收；部分出库后禁止直接取消。
3. **P0 固定资产最小台账**：资产卡、使用人/保管人、地点、状态、领用/退回/转移/报废；员工离职增加资产清点。
4. **P1 控制点**：期初、冲销、质量转换、退货纳入复核；订单增加 completed 状态并禁止履行后取消。
5. **P2 追溯与计价**：批次/库位/效期、收货质检与附件、发货回单、移动平均成本。
